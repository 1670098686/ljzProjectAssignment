## 实现细节

### 1. 启动优化工具类 (StartupOptimizer.kt)

#### 冷启动优化
```kotlin
fun optimizeColdStart(activity: Activity, criticalTask: () -> Unit, nonCriticalTask: () -> Unit) {
    // 立即执行关键任务
    criticalTask()
    
    // 延迟执行非关键任务（等待UI渲染完成）
    activity.window.decorView.post {
        nonCriticalTask()
    }
}
```

#### 热启动优化
```kotlin
fun preloadDataForWarmStart(preloadTasks: List<() -> Unit>) {
    ioScope.launch {
        preloadTasks.forEach { task ->
            try {
                task()
            } catch (e: Exception) {
                // 预加载失败不影响应用启动
                e.printStackTrace()
            }
        }
    }
}
```

#### 启动性能监控
```kotlin
class StartupMonitor {
    private var appCreateTime: Long = 0
    private var firstActivityCreateTime: Long = 0
    private var firstActivityVisibleTime: Long = 0
    
    fun onApplicationCreate() { appCreateTime = System.currentTimeMillis() }
    fun onFirstActivityCreate() { firstActivityCreateTime = System.currentTimeMillis() }
    fun onFirstActivityVisible() { firstActivityVisibleTime = System.currentTimeMillis() }
}
```

### 2. Application类优化 (CampusTradeApplication.kt)

#### 初始化流程优化
```kotlin
override fun onCreate() {
    super.onCreate()
    
    // 记录应用启动时间
    StartupOptimizer.monitor.onApplicationCreate()
    
    // 在主线程执行关键配置
    initCriticalConfig()
    
    // 在后台线程执行非关键初始化
    CoroutineScope(Dispatchers.IO).launch {
        initBackgroundConfig()
        preloadCommonData()
        
        // 执行预加载任务
        StartupOptimizer.executePreloadTasks()
    }
}
```

#### 配置分离
- **关键配置**：立即在主线程执行
- **非关键配置**：延迟在后台线程执行

### 3. 启动Activity优化 (LoginActivity.kt)

#### 启动流程优化
```kotlin
override fun onCreate(savedInstanceState: Bundle?) {
    super.onCreate(savedInstanceState)
    
    // 记录Activity创建时间
    StartupOptimizer.monitor.onFirstActivityCreate()
    
    // 使用启动优化工具
    StartupOptimizer.optimizeColdStart(
        activity = this,
        criticalTask = {
            binding = ActivityLoginBinding.inflate(layoutInflater)
            setContentView(binding.root)
            setupCriticalUI()
        },
        nonCriticalTask = {
            setupNonCriticalUI()
            setupObservers()
        }
    )
}
```

#### UI初始化分离
```kotlin
/**
 * 设置关键UI（立即执行）
 * 只设置用户立即需要交互的UI组件
 */
private fun setupCriticalUI() {
    // 设置登录按钮点击事件（用户立即需要）
    binding.btnLogin.setOnClickListener {
        val username = binding.etUsername.text.toString()
        val password = binding.etPassword.text.toString()
        viewModel.login(username, password)
    }
    
    // 设置注册跳转（用户立即需要）
    binding.tvRegister.setOnClickListener {
        val intent = Intent(this, RegisterActivity::class.java)
        startActivity(intent)
    }
}

/**
 * 设置非关键UI（延迟执行）
 * 延迟设置不需要立即使用的UI组件
 */
private fun setupNonCriticalUI() {
    // 设置输入框监听（可以延迟设置）
    binding.etUsername.doAfterTextChanged {
        viewModel.onUsernameChanged(it.toString())
    }
    
    binding.etPassword.doAfterTextChanged {
        viewModel.onPasswordChanged(it.toString())
    }
}
```
